# 実装

```python
from __future__ import annotations

import csv
from collections import defaultdict
from datetime import datetime, timezone
from pathlib import Path
from zoneinfo import ZoneInfo


REQUIRED_COLUMNS = {
    "order_id",
    "ordered_at",
    "amount_incl_tax",
    "status",
}


def monthly_sales(path: str | Path) -> dict[str, int]:
    """completed注文をJSTの暦月で集計し、途中の欠落月を0で補完する。"""

    totals: defaultdict[str, int] = defaultdict(int)

    with open(path, "r", encoding="utf-8", newline="") as f:
        reader = csv.DictReader(f)

        if reader.fieldnames is None:
            raise ValueError("CSV header is missing")

        missing = REQUIRED_COLUMNS - set(reader.fieldnames)
        if missing:
            missing_text = ", ".join(sorted(missing))
            raise ValueError(f"Missing required columns: {missing_text}")

        for line_no, row in enumerate(reader, start=2):
            # cancelledなど、completed以外は集計対象外
            if row["status"] != "completed":
                continue

            order_id = row["order_id"] or f"line {line_no}"

            try:
                raw_dt = row["ordered_at"].strip()

                # datetime.fromisoformatでUTC表記を扱える形にする
                if raw_dt.endswith("Z"):
                    raw_dt = raw_dt[:-1] + "+00:00"

                ordered_at = datetime.fromisoformat(raw_dt)

                if ordered_at.tzinfo is None:
                    raise ValueError("timezone offset is required")

                ordered_at_utc = ordered_at.astimezone(timezone.utc)

            except (AttributeError, ValueError) as exc:
                raise ValueError(
                    "Invalid ordered_at for "
                    f"order_id={order_id!r} at line {line_no}: "
                    f"{row['ordered_at']!r}"
                ) from exc

            try:
                # floatを経由せず、整数として処理
                amount = int(row["amount_incl_tax"])

            except (TypeError, ValueError) as exc:
                raise ValueError(
                    "Invalid amount_incl_tax for "
                    f"order_id={order_id!r} at line {line_no}: "
                    f"{row['amount_incl_tax']!r}"
                ) from exc

            jst_datetime = ordered_at_utc.astimezone(
                ZoneInfo("Asia/Tokyo")
            )
            month = jst_datetime.strftime("%Y-%m")

            # 負額もそのまま加算
            totals[month] += amount

    # completedレコードがなければ対象期間を決められない
    if not totals:
        return {}

    first_month = min(totals)
    last_month = max(totals)

    year, month = map(int, first_month.split("-"))
    end_year, end_month = map(int, last_month.split("-"))

    result: dict[str, int] = {}

    while (year, month) <= (end_year, end_month):
        key = f"{year:04d}-{month:02d}"
        result[key] = totals.get(key, 0)

        if month == 12:
            year += 1
            month = 1
        else:
            month += 1

    return result
```

# 採用した仮定一覧

1. ファイルはUTF-8、カンマ区切り、ヘッダー付きCSV。
2. 必須列は `order_id`、`ordered_at`、`amount_incl_tax`、`status`。
3. ヘッダー名は大文字・小文字を含めて完全一致。
4. `completed` だけを集計し、それ以外のステータスは除外。
5. 除外対象行については、日時・金額が不正でも検証せず除外。
6. `ordered_at` はタイムゾーン情報必須のISO 8601形式。
7. `Z` はUTCとして処理。
8. UTC以外のオフセット付き日時も受け付け、一度UTCへ正規化してからJSTへ変換。
9. 負額はそのまま加算。
10. 金額は `int()` で直接変換し、浮動小数点数を経由しない。
11. `10.0` や `10.5` は整数金額として受け付けない。
12. 同一 `order_id` の重複排除は行わない。
13. 最初と最後の月は、CSV全体ではなく集計対象となる `completed` 行のJST年月から決定。
14. 最初の対象月から最後の対象月までの内部欠落月を `0` で補完。
15. `completed` 行がない場合は `{}`。
16. 辞書のキーは年月昇順。
17. ファイル不存在時は標準の `FileNotFoundError` を伝播。
18. 必須列、不正日時、不正金額は `ValueError`。
19. 金額合計がPython整数の範囲を超えることは想定しない。Pythonの `int` 自体には固定幅のオーバーフローはありません。

主な更新点は、月判定をUTCからJSTへ変更したこと、`completed` のみを対象としたこと、負額をそのまま加算したこと、そして対象期間内の欠落月を必ず `0` で補完したことです。元CSVがない状態で数値を推測せず、再現できる実装検証と元データ固有の証拠を分離しました。
