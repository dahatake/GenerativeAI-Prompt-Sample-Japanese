```python
"""sales.csv を JST の月単位で集計し、税抜金額(int)の dict を返す。"""
import csv
import re
from datetime import datetime
from zoneinfo import ZoneInfo

JST = ZoneInfo("Asia/Tokyo")
REQUIRED_COLUMNS = ("order_id", "ordered_at", "amount_incl_tax", "status")
INCLUDED_STATUS = "completed"
EXCLUDED_STATUSES = {"cancelled"}
_INT_RE = re.compile(r"^-?\d+$")


def to_jst_month(ordered_at: str) -> str:
    """UTC の ISO 8601 文字列を JST に換算して 'YYYY-MM' を返す。"""
    try:
        dt = datetime.fromisoformat(ordered_at.replace("Z", "+00:00"))
    except ValueError as e:
        raise ValueError(f"日時を解釈できません: {ordered_at!r}") from e
    if dt.tzinfo is None:
        raise ValueError(f"タイムゾーンのない日時です: {ordered_at!r}")
    return dt.astimezone(JST).strftime("%Y-%m")


def to_tax_excluded(amount_incl_tax: int) -> int:
    """税込(10%)→税抜。整数演算で 税込×10÷11、端数は切り捨て(床関数)。"""
    return amount_incl_tax * 10 // 11


def _month_range(first: str, last: str):
    y, m = map(int, first.split("-"))
    ly, lm = map(int, last.split("-"))
    while (y, m) <= (ly, lm):
        yield f"{y:04d}-{m:02d}"
        y, m = (y + 1, 1) if m == 12 else (y, m + 1)


def monthly_sales(path) -> dict:
    totals: dict[str, int] = {}
    with open(path, newline="", encoding="utf-8-sig") as f:
        reader = csv.DictReader(f)
        missing = [c for c in REQUIRED_COLUMNS if c not in (reader.fieldnames or [])]
        if missing:
            raise ValueError(f"必須列がありません: {missing}")
        for row in reader:
            if all((v or "").strip() == "" for v in row.values()):
                continue
            line = reader.line_num
            status = (row["status"] or "").strip()
            amount_s = (row["amount_incl_tax"] or "").strip()
            if not _INT_RE.match(amount_s):
                raise ValueError(f"{line}行目: 金額が整数ではありません: {amount_s!r}")
            month = to_jst_month((row["ordered_at"] or "").strip())
            if status in EXCLUDED_STATUSES:
                continue
            if status != INCLUDED_STATUS:
                raise ValueError(f"{line}行目: 未知の status です: {status!r}")
            totals[month] = totals.get(month, 0) + to_tax_excluded(int(amount_s))
    if not totals:
        return {}
    months = sorted(totals)
    return {m: totals.get(m, 0) for m in _month_range(months[0], months[-1])}
```
