```python
"""sales.csv を日本時間の月ごとに集計し、税抜金額を返す。"""
import csv
from collections import defaultdict
from datetime import datetime, timedelta, timezone

JST = timezone(timedelta(hours=9))  # JST はサマータイムがないので固定オフセットで扱う
REQUIRED_COLUMNS = {"order_id", "ordered_at", "amount_incl_tax", "status"}
INCLUDED_STATUS = "completed"
EXCLUDED_STATUSES = {"cancelled"}


def _to_tax_excluded(amount_incl_tax: int) -> int:
    """税込(10%)から税抜へ変換(÷1.1)。float は使わず、0 方向に切り捨てる。"""
    sign = -1 if amount_incl_tax < 0 else 1
    return sign * (abs(amount_incl_tax) * 10 // 11)


def _parse_utc(value: str, order_id: str) -> datetime:
    try:
        dt = datetime.fromisoformat(value.strip().replace("Z", "+00:00"))
    except ValueError as e:
        raise ValueError(f"{order_id}: ordered_at を解釈できません: {value!r}") from e
    if dt.tzinfo is None:
        raise ValueError(f"{order_id}: ordered_at にタイムゾーンがありません: {value!r}")
    return dt


def _month_range(first: str, last: str):
    y, m = map(int, first.split("-"))
    ly, lm = map(int, last.split("-"))
    while (y, m) <= (ly, lm):
        yield f"{y:04d}-{m:02d}"
        y, m = (y + 1, 1) if m == 12 else (y, m + 1)


def monthly_sales(path) -> dict:
    """{"YYYY-MM": 税抜金額(int)} を返す。最初の月から最後の月まで、注文がない月は 0。"""
    totals = defaultdict(int)
    with open(path, encoding="utf-8-sig", newline="") as f:
        reader = csv.DictReader(f)  # 空行は DictReader が読み飛ばす
        missing = REQUIRED_COLUMNS - set(reader.fieldnames or [])
        if missing:
            raise ValueError(f"必要な列がありません: {sorted(missing)}")
        for row in reader:
            order_id = (row["order_id"] or "").strip()
            status = (row["status"] or "").strip()
            if status in EXCLUDED_STATUSES:
                continue
            if status != INCLUDED_STATUS:
                raise ValueError(f"{order_id}: 想定外の status です: {status!r}")
            try:
                amount = int((row["amount_incl_tax"] or "").strip())
            except ValueError as e:
                raise ValueError(f"{order_id}: 金額を整数として解釈できません: "
                                 f"{row['amount_incl_tax']!r}") from e
            month = _parse_utc(row["ordered_at"] or "", order_id).astimezone(JST).strftime("%Y-%m")
            totals[month] += _to_tax_excluded(amount)
    if not totals:
        return {}
    keys = sorted(totals)
    return {m: totals.get(m, 0) for m in _month_range(keys[0], keys[-1])}
```

```python
import os, re, tempfile, unittest
from monthly_sales import monthly_sales

HEADER = "order_id,ordered_at,amount_incl_tax,status\n"
EXPECTED = {"2026-01": 15000, "2026-02": 10000, "2026-03": 0, "2026-04": 17000, "2026-05": 7000}

def write(text, encoding="utf-8"):
    fd, p = tempfile.mkstemp(suffix=".csv"); os.close(fd)
    with open(p, "w", encoding=encoding, newline="") as f: f.write(text)
    return p

class DoD(unittest.TestCase):

        print("\n  DoD1 結果:", r)
        self.assertEqual(r, EXPECTED)
        self.assertEqual(list(r), sorted(r))

    def test_dod2_jst_boundary(self):
        p = write(HEADER + "B1,2026-01-31T14:59:59Z,1100,completed\n"
                           "B2,2026-01-31T15:00:00Z,2200,completed\n")
        r = monthly_sales(p); print("\n  DoD2 境界:", r)
        self.assertEqual(r, {"2026-01": 1000, "2026-02": 2000})

    def test_dod2_real_rows(self):
```
